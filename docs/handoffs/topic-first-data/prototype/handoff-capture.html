fct_gl_transactions.sql incremental · edit to override 
select
    t.transaction_id                      as gl_transaction_key,
    d.date_key, a.account_key,
    g.gl_account_key, j.journal_key,
    t.amount_dc                           as amount,
    t.currency, t.description
from  exactonline.transaction_lines t
left join dim_date        d on d.full_date    = t.entry_date
left join dim_accounts    a on a.account_id   = t.account_id
left join dim_gl_accounts g on g.gl_code      = t.gl_account
left join dim_journals    j on j.journal_code = t.journal
where t.deleted is null
