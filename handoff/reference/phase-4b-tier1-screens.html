06SQL0.4sRun⋯
-- top customers that moved by ≥ €50K, Q3 vs Q2
SELECT c.name, sum(q3.amt) AS q3_rev, sum(q2.amt) AS q2_rev
FROM dim_customer c JOIN fact_sales q3 ON q3.customer_id = c.id
WHERE q3.quarter = 'Q3 2025'
GROUP BY c.name HAVING abs(q3_rev - q2_rev) >= 50000
ORDER BY abs(q3_rev - q2_rev) DESC
